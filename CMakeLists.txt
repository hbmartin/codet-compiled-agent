cmake_minimum_required(VERSION 3.20)
project(codet VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CODET_BUILD_TESTS "Build hermetic tests" ON)
option(CODET_FOUNDATION_ONLY "Build the test foundation before application modules" OFF)
option(CODET_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
set(CODET_DEPENDENCY_ROOT "${CMAKE_SOURCE_DIR}/.deps" CACHE PATH "Offline dependency sources")
foreach(dep googletest json tomlplusplus json-schema-validator)
  if(NOT EXISTS "${CODET_DEPENDENCY_ROOT}/${dep}/CMakeLists.txt")
    message(FATAL_ERROR "Missing pinned ${dep}; run python3 scripts/fetch-deps.py first")
  endif()
endforeach()
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
set(TOMLPLUSPLUS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(JSON_VALIDATOR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(JSON_VALIDATOR_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(JSON_VALIDATOR_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${CODET_DEPENDENCY_ROOT}/json" "deps/json" EXCLUDE_FROM_ALL)
add_subdirectory("${CODET_DEPENDENCY_ROOT}/tomlplusplus" "deps/toml" EXCLUDE_FROM_ALL)
add_subdirectory("${CODET_DEPENDENCY_ROOT}/json-schema-validator" "deps/schema" EXCLUDE_FROM_ALL)
if(CODET_FOUNDATION_ONLY)
  enable_testing()
  set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
  add_subdirectory("${CODET_DEPENDENCY_ROOT}/googletest" "deps/gtest" EXCLUDE_FROM_ALL)
  add_executable(codet_foundation_tests tests/foundation_test.cpp src/text_compat.c)
  target_include_directories(codet_foundation_tests PRIVATE include)
  target_link_libraries(codet_foundation_tests PRIVATE nlohmann_json::nlohmann_json GTest::gmock_main)
  add_test(NAME codet.foundation COMMAND codet_foundation_tests)
  return()
endif()
find_package(CURL REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(ZLIB REQUIRED)
find_package(SQLite3 REQUIRED)
if(TARGET SQLite3::SQLite3)
  set(CODET_SQLITE_TARGET SQLite3::SQLite3)
else()
  set(CODET_SQLITE_TARGET SQLite::SQLite3)
endif()
find_package(Curses REQUIRED)
find_package(Threads REQUIRED)
find_path(EDIT_INCLUDE_DIR histedit.h)
find_library(EDIT_LIBRARY edit REQUIRED)
if(NOT EDIT_INCLUDE_DIR)
  message(FATAL_ERROR "libedit headers (histedit.h) are required")
endif()
add_library(codet_text STATIC src/text_compat.c)
target_include_directories(codet_text PUBLIC include)
add_library(codet_core STATIC
  src/modern/platform.cpp src/modern/config.cpp src/modern/provider.cpp
  src/modern/state.cpp src/modern/hooks.cpp src/modern/patch.cpp src/modern/tools.cpp src/modern/agent.cpp src/modern/cli.cpp)
target_include_directories(codet_core PUBLIC include PRIVATE ${CURSES_INCLUDE_DIRS} ${EDIT_INCLUDE_DIR})
target_link_libraries(codet_core PUBLIC nlohmann_json::nlohmann_json tomlplusplus::tomlplusplus
  nlohmann_json_schema_validator::validator CURL::libcurl OpenSSL::SSL OpenSSL::Crypto ZLIB::ZLIB ${CODET_SQLITE_TARGET} Threads::Threads
  ${CURSES_LIBRARIES} ${EDIT_LIBRARY})
if(CMAKE_CROSSCOMPILING)
  target_link_libraries(codet_core PUBLIC "${_prefix}/lib/libMacportsLegacySupport.a")
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_libraries(codet_core PUBLIC util)
endif()
target_compile_options(codet_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_definitions(codet_core PUBLIC CODET_VERSION="${PROJECT_VERSION}")
if(CODET_SANITIZERS)
  if(CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "Sanitizers are host-only")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(codet_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(codet_text PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
add_executable(codet src/modern/main.cpp)
target_link_libraries(codet PRIVATE codet_core)
if(CODET_BUILD_TESTS)
  enable_testing()
  set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
  set(BUILD_GMOCK ON CACHE BOOL "" FORCE)
  add_subdirectory("${CODET_DEPENDENCY_ROOT}/googletest" "deps/gtest" EXCLUDE_FROM_ALL)
  add_executable(codet_tests tests/foundation_test.cpp tests/core_test.cpp tests/review_test.cpp tests/remaining_test.cpp)
  target_link_libraries(codet_tests PRIVATE codet_core codet_text GTest::gmock_main)
  # No discovery-time execution: the same build must work for PPC cross builds.
  if(NOT CMAKE_CROSSCOMPILING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME codet.unit COMMAND codet_tests)
    add_test(NAME codet.cli COMMAND ${CMAKE_COMMAND} -DCODET=$<TARGET_FILE:codet>
      -DTEST_ROOT=${CMAKE_BINARY_DIR}/cli-test -P ${CMAKE_SOURCE_DIR}/tests/cli_test.cmake)
    add_test(NAME codet.integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/cli_integration.py $<TARGET_FILE:codet>)
    set_tests_properties(codet.integration PROPERTIES TIMEOUT 60)
  endif()
endif()
install(TARGETS codet RUNTIME DESTINATION bin)
